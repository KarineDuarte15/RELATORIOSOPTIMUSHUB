      COALESCE(

      COALESCE(
          CALCULATE([01_Qtd_Registros_PA_Digital],
              FILTER('BASE_PGTO_MEDICO_DIGITAL',
              'BASE_PGTO_MEDICO_DIGITAL'[STATUS INTEGRAÇÃO] = "INTEGRADO")),0)

          [05_Qtd_NFE_Integrada_PA_Digital],
          [04_Liberado_Pgto_PA_Digital],
          0)
    - 03_Qtd_processos_PA_Digital:
      
      COALESCE(
          CALCULATE(
              '#MEDIDAS_PA_DIGITAL'[01_Qtd_Registros_PA_Digital],

                  'BASE_PGTO_MEDICO_DIGITAL',
                  'BASE_PGTO_MEDICO_DIGITAL'[STATUS PROCESSO] <> "SEM ATENDIMENTO"
          )
      ),0
      )
