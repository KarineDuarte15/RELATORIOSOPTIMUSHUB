Tabela: #MEDIDAS
  Medidas DAX (Regras de Negócio):
    - Qt Cons. Total:
      
      VAR MaxDataOperacaoPorMes = 
              MAX(fTransferencia[DT_OPERACAO])
      RETURN
          CALCULATE(
              SUM(fTransferencia[QT_TOTAL]),
              fTransferencia[DT_OPERACAO] = MaxDataOperacaoPorMes
          )
      
    - Tx Transf.:
      DIVIDE( [Qt Cons. Transf] , [Qt Cons. Total] )
    - Qt Cons. Transf:
      
      VAR TabelaAgregada = 
          SUMMARIZE(
              fTransferencia,
              fTransferencia[REF_CONSULTA],
              fTransferencia[DT_OPERACAO],
              "TotalTransf", SUM(fTransferencia[QT_TRANSF])
          )
      VAR MaxDataOperacaoPorMes = 
          MAXX(
              FILTER(TabelaAgregada, fTransferencia[REF_CONSULTA] = SELECTEDVALUE(fTransferencia[REF_CONSULTA])),
              fTransferencia[DT_OPERACAO]
          )
      RETURN 
          MAXX(
              FILTER(TabelaAgregada, fTransferencia[DT_OPERACAO] = MaxDataOperacaoPorMes),
              [TotalTransf]
          )
    - Consulta Transf. Dia:
      
      VAR Total = SUM(fTransferencia[QT_TRANSF])
      RETURN 
      IF(Total > 0, Total, BLANK())
      
    - Qt Registros:
      COUNTROWS( fTransferencia )
    - DIAS ABERTO:
      SUM('fDiasAbertos'[Dias Abertos])
    - DataMaxima:
      MAX(fDiasAbertos[DT_FIM])
    - Total Transf em Aberto:
      
      CALCULATE(
          COUNTROWS( fDiasAbertos ) , 
          FILTER(
              fDiasAbertos, 
              fDiasAbertos[STATUS AGENDA] = 1 )
      )
    - Total Transf em Fechado:
      
      CALCULATE(
          COUNTROWS( fDiasAbertos ) , 
          FILTER(
              fDiasAbertos, 
              fDiasAbertos[STATUS AGENDA] = 0 )
      )
    - _Msg Boas Vindas:
      
      VAR vUsuario = LEFT(USERPRINCIPALNAME(), FIND("@", USERPRINCIPALNAME()) - 1)
      VAR vHorarioAtual = NOW() - TIME(3, 0, 0) // Subtrai 3 horas para ajustar ao fuso horário de Brasília (UTC -3)
      VAR vSaudacao = 
          SWITCH(
              TRUE(),
              HOUR(vHorarioAtual) < 12, ", bom dia. Seja bem-vindo(a)!",
              HOUR(vHorarioAtual) < 18, ", boa tarde. Seja bem-vindo(a)!",
              HOUR(vHorarioAtual) <= 23, ", boa noite. Seja bem-vindo(a)!"
          )
      RETURN
      "Olá, " & vUsuario & vSaudacao
      
    - Transfencia:
      sum(fTransfCalendario[QT_TRANSF])
    - Max_Dia:
      MAX( fTransferencia[DT_OPERACAO])

Tabela: dAtualizacao
  Colunas:
    - dAtualizacao [dateTime]
  Medidas DAX (Regras de Negócio):
    - Última Atualização:
      
      VAR DataRelatorio =
          SUM(dAtualizacao[dAtualizacao])
      
      VAR DataDados =
          [DataMaxima]
      
      VAR DifDias =
          DATEDIFF(
              INT(DataDados),
              INT(DataRelatorio),
              DAY
          )
      
      RETURN
      "Última atualização do relatório: "
          & FORMAT(DataRelatorio, "dd/MM/yyyy HH:mm")
          & " Última atualização dos dados: "
          & FORMAT(DataDados, "dd/MM/yyyy")
          & " | (D-" & DifDias & ")"
