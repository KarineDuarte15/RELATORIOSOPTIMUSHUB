Tabela: MEDIDAS
  Colunas:
    - Column [int64]
  Medidas DAX (Regras de Negócio):
    - QTD_SAC_TOTAL:
      IF(
          SUM(FATO_UNIFICADA[SAC_GERAL]) = BLANK(), 0,
           SUM(FATO_UNIFICADA[SAC_GERAL])
           ) 
    - QTD_SAC_REDE_CRED:
      
      IF(
          SUM(FATO_UNIFICADA[SAC_RC]) = BLANK(),0,
          SUM(FATO_UNIFICADA[SAC_RC])  
      )
      
    - QTD_SAC_REDE_PROPRIA:
      
      IF(
          SUM(FATO_UNIFICADA[SAC_RP]) = BLANK(),0,
          SUM(FATO_UNIFICADA[SAC_RP])
      )
      
    - (%) SAC_RPR:
      IFERROR(DIVIDE([QTD_SAC_REDE_PROPRIA],[QTD_SAC_TOTAL]),0)
    - (%) SAC_CRED:
      IFERROR(DIVIDE([QTD_SAC_REDE_CRED],[QTD_SAC_TOTAL]),0)
    - TESTE:
      
      
      VAR DIA = DAY(EOMONTH(TODAY()-1,0))
      VAR MES = MONTH(EOMONTH(TODAY()-1,0))
      
      RETURN
      
      MES
    - Msg Boas Vindas:
      
      VAR vUsuario = LEFT(USERPRINCIPALNAME(), FIND("@", USERPRINCIPALNAME()) - 1)
      VAR vHorarioAtual = NOW() - TIME(3, 0, 0) // Subtrai 3 horas para ajustar ao fuso horário de Brasília (UTC -3)
      VAR vSaudacao = 
          SWITCH(
              TRUE(),
              HOUR(vHorarioAtual) < 12, ", bom dia. Seja bem-vindo(a)!",
              HOUR(vHorarioAtual) < 18, ", boa tarde. Seja bem-vindo(a)!",
              HOUR(vHorarioAtual) <= 23, ", boa noite. Seja bem-vindo(a)!"
          )
      RETURN
      "Olá, " & vUsuario & vSaudacao
    - Data e Hora Atual:
      
      NOW() - TIME(3,0,0)
    - % SAC:
      
      VAR TotalMes =
          CALCULATE(
              [QTD_SAC_TOTAL],
              ALL('FATO_UNIFICADA'[TIPO_REDE_ATENDIMENTO])
          )
      RETURN
      DIVIDE(
          [QTD_SAC_TOTAL],
          TotalMes,
          0
      )
