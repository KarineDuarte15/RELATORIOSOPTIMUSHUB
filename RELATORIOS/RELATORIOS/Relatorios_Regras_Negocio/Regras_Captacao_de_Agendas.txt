  Colunas:
    - Ano [int64]
    - Mes [int64]
    - MES_REF [string]
    - MES_ANO [string]
    - NM_MES [string]
    - TRIMESTRE [string]
    - ORDEM_MES [int64]
    - DATA_MES [dateTime]

Tabela: _Medidas
  Colunas:
    - Column [int64]
  Medidas DAX (Regras de Negócio):
    - QT DISP:
      
      COALESCE(SUM(fMonitoramento[QT_MES_ATUAL]), 0)
      
    - QT MEDIANA 6M:
      
      COALESCE(SUM(fMonitoramento[QT_MEDIANA_6M]), 0)
    - QT MAX 6M:
      
      COALESCE(SUM(fMonitoramento[QT_MAX_6M]), 0)
    - DESVIO MEDIANA:
      
      DIVIDE(
          [QT DISP] - [QT MEDIANA 6M],
          [QT MEDIANA 6M],
          0
      )
    - DESVIO MAX:
      
      DIVIDE(
          [QT DISP] - [QT MAX 6M],
          [QT MAX 6M],
          0
      )
    - RISCO MAX:
      
      IF(
          ISINSCOPE(fMonitoramento[NM_MEDICO]) || ISINSCOPE(fMonitoramento[CD_LOCAL]),
          VAR MesAtual = [QT DISP]
          VAR Max6M    = [QT MAX 6M]
          VAR Pct      = DIVIDE(MesAtual, Max6M, 0)
          RETURN
              IF(
                  Max6M = 0, "Sem histórico",
                  IF(Pct >= 0.90, "🟢 Agenda Entregue",
                  IF(Pct >= 0.75, "🟡 Entrega Parcial",
                                  "🔴 Não Entregue"))
              ),
          BLANK()
      )
    - QT MEDICOS:
      
      COUNTROWS(
          SUMMARIZE(
              fMonitoramento,
              fMonitoramento[CD_MEDICO],
              fMonitoramento[CD_ESPECIALIDADE],
              fMonitoramento[CD_LOCAL]
          )
      )
    - QT MEDICOS RECORRENTES:
      
      COUNTROWS(
          SUMMARIZE(
              FILTER(
                  fMonitoramento,
                  fMonitoramento[RECORRENCIA] = "Recorrente"
              ),
              fMonitoramento[CD_MEDICO],
              fMonitoramento[CD_ESPECIALIDADE],
              fMonitoramento[CD_LOCAL]
          )
      )
    - Card 1 Medicos HTML:
      
      VAR TotalMedicos = [QT MEDICOS]
      VAR Recorrentes  = [QT MEDICOS RECORRENTES]
      VAR PctRec       = DIVIDE(Recorrentes, TotalMedicos, 0)
      
      VAR MesAtual =
          SELECTEDVALUE(fMonitoramento[MES_AGENDA])
      
      VAR MesAnterior =
          CALCULATE(
              MAX(fMonitoramento[MES_AGENDA]),
              FILTER(
                  ALL(fMonitoramento[MES_AGENDA]),
                  fMonitoramento[MES_AGENDA] < MesAtual
              )
          )
      
      VAR TotalAnterior =
          IF(ISBLANK(MesAnterior), BLANK(), CALCULATE([QT MEDICOS], fMonitoramento[MES_AGENDA] = MesAnterior))
      
      VAR RecAnterior =
          IF(ISBLANK(MesAnterior), BLANK(), CALCULATE([QT MEDICOS RECORRENTES], fMonitoramento[MES_AGENDA] = MesAnterior))
      
      VAR PctRecAnterior =
          IF(ISBLANK(MesAnterior), BLANK(), DIVIDE(RecAnterior, TotalAnterior, 0))
      
      VAR DeltaTotal =
          IF(ISBLANK(TotalAnterior), BLANK(), DIVIDE(TotalMedicos - TotalAnterior, TotalAnterior, 0))
      
      VAR DeltaTotalFmt =
          IF(
              ISBLANK(DeltaTotal), "Mês base",
              IF(DeltaTotal >= 0,
                  "▲ +" & FORMAT(ABS(DeltaTotal), "0.0%"),
                  "▼ "  & FORMAT(ABS(DeltaTotal), "0.0%"))
          )
      
      VAR CorDeltaTotal = IF(ISBLANK(DeltaTotal) || DeltaTotal >= 0, "#14643c", "#c0392b")
      VAR BgDeltaTotal  = IF(ISBLANK(DeltaTotal) || DeltaTotal >= 0, "#e8f5ee", "#fdecea")
      
      VAR DeltaRec =
          IF(ISBLANK(PctRecAnterior), BLANK(), PctRec - PctRecAnterior)
      
      VAR DeltaRecFmt =
          IF(
              ISBLANK(DeltaRec), "Mês base",
              IF(DeltaRec >= 0,
                  "▲ +" & FORMAT(ABS(DeltaRec) * 100, "0.0") & " p.p.",
                  "▼ "  & FORMAT(ABS(DeltaRec) * 100, "0.0") & " p.p.")
          )
      
      VAR CorDeltaRec = IF(ISBLANK(DeltaRec) || DeltaRec >= 0, "#14643c", "#c0392b")
      VAR BgDeltaRec  = IF(ISBLANK(DeltaRec) || DeltaRec >= 0, "#e8f5ee", "#fdecea")
      
      RETURN
      "<div style='font-family:Segoe UI,sans-serif;background:#fff;width:404.83px;height:223px;padding:16px 20px 14px;display:flex;flex-direction:column;justify-content:space-between;border-top:2px solid #1a5fa8;border-radius:9px;box-shadow:0 1px 2px rgba(0,0,0,.05),0 2px 6px rgba(0,0,0,.04);box-sizing:border-box;'>
      
        <div style='display:flex;justify-content:space-between;align-items:flex-start;'>
          <span style='font-size:10px;font-weight:700;letter-spacing:.9px;text-transform:uppercase;color:#b0bac8;'>Cobertura de Médicos</span>
          <span style='font-size:15px;opacity:.5;'>👨‍⚕️</span>
        </div>
      
        <div>
          <div style='display:flex;align-items:baseline;gap:10px;'>
            <span style='font-size:38px;font-weight:800;letter-spacing:-0.5px;line-height:1;color:#1a5fa8;'>" & FORMAT(TotalMedicos, "#,##0") & "</span>
            <span style='font-size:14px;color:#6b7a99;font-weight:500;'>médicos elegíveis</span>
            <span style='display:inline-flex;align-items:center;font-size:11px;font-weight:700;padding:3px 8px;border-radius:10px;background:" & BgDeltaTotal & ";color:" & CorDeltaTotal & ";margin-left:auto;'>" & DeltaTotalFmt & "</span>
          </div>
      
          <div style='display:flex;align-items:baseline;gap:10px;margin-top:8px;'>
            <span style='font-size:28px;font-weight:800;letter-spacing:-0.5px;line-height:1;color:#14643c;'>" & FORMAT(PctRec, "0.0%") & "</span>
            <span style='font-size:13px;color:#6b7a99;font-weight:500;'>são recorrentes (" & FORMAT(Recorrentes, "#,##0") & ")</span>
            <span style='display:inline-flex;align-items:center;font-size:11px;font-weight:700;padding:3px 8px;border-radius:10px;background:" & BgDeltaRec & ";color:" & CorDeltaRec & ";margin-left:auto;'>" & DeltaRecFmt & "</span>
          </div>
        </div>
      
        <div style='font-size:11px;color:#9aa6b8;line-height:1.5;padding-top:10px;border-top:1px solid #f3f5f8;'>
          Cada combinação <b style='color:#6b7a99;'>Médico + Especialidade + Unidade</b> conta como um elegível. São <b style='color:#14643c;'>recorrentes</b> os que disponibilizaram em pelo menos 3 dos últimos 6 meses.
        </div>
      
      </div>"

      COUNTROWS(

              SUMMARIZE(
                  fMonitoramento,

              SUMMARIZE(
                  fMonitoramento,

                  fMonitoramento[CD_LOCAL],

              fMonitoramento[FL_DISP_MES_ATUAL] = "N"
          )

      VAR TotalDisp    = [QT DISP]
      VAR TotalMax     = [QT MAX 6M]
      VAR Aprov        = DIVIDE(TotalDisp, TotalMax, 0)
      VAR GapPP        = (0.90 - Aprov) * 100
      VAR MesAtual     = SELECTEDVALUE(fMonitoramento[MES_AGENDA])
      VAR MesAnterior =
          CALCULATE(
              MAX(fMonitoramento[MES_AGENDA]),
              FILTER(
                  ALL(fMonitoramento[MES_AGENDA]),
                  fMonitoramento[MES_AGENDA] < MesAtual
              )
          )
      VAR AprovAnt     = IF(
                           ISBLANK(MesAnterior), BLANK(),
                           DIVIDE(
                               CALCULATE([QT DISP],   fMonitoramento[MES_AGENDA] = MesAnterior),
                               CALCULATE([QT MAX 6M], fMonitoramento[MES_AGENDA] = MesAnterior),
                               0
                           )
                         )
      VAR Delta        = IF(ISBLANK(AprovAnt), BLANK(), Aprov - AprovAnt)
      VAR DeltaFmt     = IF(
                           ISBLANK(Delta), "Mês base",
                           IF(Delta >= 0,
                               "▲ +" & FORMAT(ABS(Delta)*100, "0.0") & " p.p.",
                               "▼ "  & FORMAT(ABS(Delta)*100, "0.0") & " p.p.")
                         )
      VAR CorDelta     = IF(ISBLANK(Delta) || Delta >= 0, "#14643c", "#c0392b")
      VAR BgDelta      = IF(ISBLANK(Delta) || Delta >= 0, "#e8f5ee",  "#fdecea")
      VAR CorBarra     = IF(Aprov >= 0.90, "#14643c", IF(Aprov >= 0.75, "#d97b00", "#c0392b"))
      VAR LargBarra    = FORMAT(MIN(Aprov * 100, 100), "0") & "%"
      VAR GapFmt       = FORMAT(GapPP, "0.0") & " p.p."
      VAR GapConsultas = MAX(0, ROUND(TotalMax * 0.9 - TotalDisp, 0))
      VAR LabelEsquerda = IF(Aprov >= 0.90, "✅ Meta atingida", "Faltam " & FORMAT(GapConsultas, "#,##0") & " consultas p/ atingir a meta")
      VAR MesMaisRecente = CALCULATE(MAX(fMonitoramento[MES_AGENDA]), ALL(fMonitoramento[MES_AGENDA]))
      VAR IsAgo        = MesAtual = MesMaisRecente
      VAR BadgeAgo     = "<span style='display:inline-flex;align-items:center;font-size:12px;font-weight:700;padding:3px 9px;border-radius:10px;background:#fff8e6;color:#7a5c00;'>⚠️ Em confecção</span>"
      VAR BadgeDelta   = "<span style='display:inline-flex;align-items:center;font-size:12px;font-weight:700;padding:3px 9px;border-radius:10px;background:" & BgDelta & ";color:" & CorDelta & ";'>" & DeltaFmt & " vs mês anterior</span>"
      
      RETURN
      "<div style='font-family:Segoe UI,sans-serif;background:#fff;width:404.83px;height:223px;padding:16px 20px 14px;display:flex;flex-direction:column;justify-content:space-between;border-top:2px solid #d97b00;border-radius:9px;box-shadow:0 1px 2px rgba(0,0,0,.05),0 2px 6px rgba(0,0,0,.04);box-sizing:border-box;'>
        <div style='display:flex;justify-content:space-between;align-items:flex-start;'>
          <span style='font-size:10px;font-weight:700;letter-spacing:.9px;text-transform:uppercase;color:#b0bac8;'>Utilização da Capacidade</span>
          <span style='font-size:15px;opacity:.5;'>📊</span>
        </div>
        <div>
          <div style='font-size:42px;font-weight:800;letter-spacing:-0.5px;line-height:1;color:#c07b0a;'>" & FORMAT(Aprov, "0.0%") & "</div>
          <div style='font-size:13px;color:#8694ab;margin-top:3px;'>" & FORMAT(TotalDisp, "#,##0") & " de " & FORMAT(TotalMax, "#,##0") & " consultas</div>
          <div style='margin-top:6px;'>" & IF(IsAgo, BadgeAgo, BadgeDelta) & "</div>
          <div style='margin:12px 0 2px;'>
            <div style='display:flex;justify-content:space-between;font-size:11px;color:#8694ab;margin-bottom:4px;'><span>" & LabelEsquerda & "</span><span>Meta 90%</span></div>
            <div style='height:5px;background:#f0f3f8;border-radius:4px;overflow:hidden;'><div style='width:" & LargBarra & ";height:100%;background:" & CorBarra & ";border-radius:4px;'></div></div>
          </div>
        </div>
        <div style='display:flex;justify-content:flex-start;align-items:center;font-size:12px;padding-top:8px;border-top:1px solid #f3f5f8;'>
          <span style='color:#9aa6b8;'>QT Disp / QT Máx · Visão T22J2 </span>
        </div>
      </div>"
    - QT SEM CONTROLE NOMINAL:
      
      COUNTROWS(
          FILTER(
              SUMMARIZE(
                  fMonitoramento,
                  fMonitoramento[MES_AGENDA],
                  fMonitoramento[CD_MEDICO],
                  fMonitoramento[CD_ESPECIALIDADE],

                  fMonitoramento[STATUS]

    - Card 6 Aderencia HTML:
      
      VAR QtdCidades        = DISTINCTCOUNT(fMonitoramento[CIDADE])
      VAR QtdEspecialidades = DISTINCTCOUNT(fMonitoramento[DS_ESPECIALIDADE])
      VAR QtdSubRegioes     = DISTINCTCOUNT(fMonitoramento[SUB_REGIAO])
      VAR CidadeAtual        = SELECTEDVALUE(fMonitoramento[CIDADE])
      VAR EspecialidadeAtual = SELECTEDVALUE(fMonitoramento[DS_ESPECIALIDADE])
      VAR SubRegiaoAtual      = SELECTEDVALUE(fMonitoramento[SUB_REGIAO])
      
      VAR CidadeCodAtual = SELECTEDVALUE(fMonitoramento[CD_LOCALIDADE_CIDADE])
      VAR EspCodAtual    = SELECTEDVALUE(fMonitoramento[CD_ESPECIALIDADE])
      VAR MesAtual       = SELECTEDVALUE(fMonitoramento[MES_AGENDA])
      
      VAR EhSubRegiaoUnica = QtdSubRegioes = 1
      
      VAR ComboFiltrado =
          FILTER(
              FORECAST_UNIVERSO_COMPLETO,
              [CD_LOCALIDADE_CIDADE] = CidadeCodAtual &&
              [CD_ESPECIALIDADE] = EspCodAtual &&
              (ISBLANK(MesAtual) || [MES] = MesAtual) &&
              [QT_COMPRA_UNIVERSO] > 0
          )
      
      VAR ComboSemMistas =
          FILTER(
              ComboFiltrado,
              VAR CidAtual = [CD_LOCALIDADE_CIDADE]
              VAR EspComb  = [CD_ESPECIALIDADE]
              VAR MesComb  = [MES]
              VAR EhMista =
                  CALCULATE(
                      COUNTROWS(AGENDAS_MISTAS_HISTORICO),
                      FILTER(
                          ALL(AGENDAS_MISTAS_HISTORICO),
                          AGENDAS_MISTAS_HISTORICO[CD_ESPECIALIDADE] = EspComb &&
                          AGENDAS_MISTAS_HISTORICO[CD_LOCALIDADE] = CidAtual &&
                          AGENDAS_MISTAS_HISTORICO[MES_ANO] = MesComb
                      )
                  )
              RETURN
                  EhMista = 0
          )
      
      VAR TotalCombos = COUNTROWS(ComboSemMistas)
      VAR CombosOK =
          COUNTROWS(
              FILTER(
                  ComboSemMistas,
                  DIVIDE([QT_DISP_UNIVERSO], [QT_COMPRA_UNIVERSO], 0) >= 0.90
              )
          )
      
      VAR AderenciaCombo = DIVIDE(CombosOK, TotalCombos, 0) * 100
      VAR AderenciaLocal = DIVIDE([QT DISP FORECAST], [QT COMPRA], 0) * 100
      
      VAR Aderencia = IF(EhSubRegiaoUnica, AderenciaLocal, AderenciaCombo)
      
      VAR DiasZeradosPorSub =
          SUMMARIZE(
              fMonitoramento,
              fMonitoramento[SUB_REGIAO],
              "_DZ", MAX(fMonitoramento[DIAS_ZERADOS])
          )
      VAR DiasZeradosMin = MINX(DiasZeradosPorSub, [_DZ])
      VAR DiasZeradosBruto = MAX(fMonitoramento[DIAS_ZERADOS])
      
      VAR DiasZerados = IF(EhSubRegiaoUnica, DiasZeradosBruto, DiasZeradosMin)
      
      VAR Meta               = 90
      VAR EhAderente         = Aderencia >= Meta
      VAR CorStatus          = IF(EhAderente, "rgb(20,100,60)", "#c0392b")
      VAR BgStatus           = IF(EhAderente, "rgb(220,245,232)", "rgb(255,235,235)")
      VAR TextoStatus        = IF(EhAderente, "✅ Aderente", "❌ Não Aderente")
      VAR CorDias            = IF(DiasZerados = 0, "rgb(20,100,60)", IF(DiasZerados <= 7, "#e6a817", "#c0392b"))
      VAR BgDias             = IF(DiasZerados = 0, "rgb(220,245,232)", IF(DiasZerados <= 7, "rgb(255,248,220)", "rgb(255,235,235)"))
      
      VAR Rodape =
          IF(
              EhSubRegiaoUnica,
              CidadeAtual & " · " & EspecialidadeAtual & " · " & SubRegiaoAtual,
              CidadeAtual & " · " & EspecialidadeAtual & " · " &
              IF(EhAderente, "Meta de 90% atingida", CombosOK & " de " & TotalCombos & " combinações ≥ 90%")
          )
      
      RETURN
      IF(
          QtdCidades = 1 && QtdEspecialidades = 1,
          "<div style='font-family:Segoe UI,sans-serif;background:#fff;width:404.83px;height:223px;padding:16px 20px 14px;display:flex;flex-direction:column;justify-content:space-between;border-top:2px solid #1a5fa8;border-radius:9px;box-shadow:0 1px 2px rgba(0,0,0,.05),0 2px 6px rgba(0,0,0,.04);box-sizing:border-box;'>
      
            <div style='display:flex;justify-content:space-between;align-items:flex-start;'>
              <span style='font-size:10px;font-weight:700;letter-spacing:.9px;text-transform:uppercase;color:#b0bac8;'>Aderência Forecast + Dias Zerados</span>
              <span style='font-size:15px;opacity:.5;'>✅</span>
            </div>
      
            <div style='display:flex;align-items:center;width:100%;justify-content:center;flex:1;'>
              <div style='text-align:center;flex:1;'>
                <div style='font-size:11px;color:#8694ab;margin-bottom:4px;'>📊 % Aderência</div>
                <div style='font-size:38px;font-weight:800;letter-spacing:-0.5px;color:#2c3e50;line-height:1;'>" & FORMAT(Aderencia, "0.00") & "%</div>
                <div style='margin-top:6px;'><span style='font-size:12px;font-weight:700;padding:3px 9px;border-radius:10px;background:" & BgStatus & ";color:" & CorStatus & ";display:inline-block;'>" & TextoStatus & "</span></div>

              <div style='width:1px;height:64px;background:#f0f3f8;'></div>

                <div style='margin-top:6px;'><span style='font-size:12px;font-weight:700;padding:3px 9px;border-radius:10px;background:" & BgDias & ";color:" & CorDias & ";display:inline-block;'>dias sem vagas</span></div>
              </div>
            </div>
      
            <div style='text-align:center;font-size:12px;color:#9aa6b8;padding-top:8px;border-top:1px solid #f3f5f8;'>" & Rodape & "</div>

      "<div style='font-family:Segoe UI,sans-serif;background:transparent;padding:32px 40px;width:100%;max-width:900px;box-sizing:border-box;color:#2c3e50;line-height:1.6'>

      <div style='margin-bottom:24px'>

O escopo contempla apenas agendas nominais <b>HAPCLINICA</b> e <b>CAPTADO</b>, com especialidades ativas no forecast de compra.</div>

        <div style='font-size:16px;color:#444'>Para cada mês de agenda (<b>MES_AGENDA</b>), utilizamos uma janela de <b>6 meses</b>: os 5 meses anteriores ao mês alvo (para calcular mediana e máximo histórico) mais o próprio mês alvo (para medir a disponibilização atual).</div>

        <div style='font-size:16px;color:#444'>Um médico é considerado elegível se disponibilizou agenda em qualquer um dos 6 meses da janela, seja no histórico (antes da confecção) ou durante o período de monitoramento (após a confecção).</div>

        <div style='font-size:16px;color:#444;margin-bottom:14px'><b>DESVIO MEDIANA / DESVIO MAX</b> — variação percentual do mês atual em relação à mediana/máximo histórico</div>
