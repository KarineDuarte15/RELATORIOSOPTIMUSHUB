          SWITCH(
              TRUE(),
              HOUR(vHorarioAtual) < 12, ", bom dia. Seja bem-vindo(a)!",

              HOUR(vHorarioAtual) <= 23, ", boa noite. Seja bem-vindo(a)!"
          )

    - (Ajusta 01.04) _Qt Zerados Cidade:
      
      VAR _FiltroClassificacao =
          VALUES(dClassificacao[CD_CLASSIFICACAO])
      
      VAR _TabelaCidadeEspecialidade =
          CALCULATETABLE(
              SUMMARIZECOLUMNS(
                  dCidade[NM_CIDADE],
                  dEspecialidade[DS_ESPECIALIDADE],
                  dCalendario[Date],
                  "QtVagas", [Qt Cons+Enc Manhã]
              ),
              TREATAS(_FiltroClassificacao, fVagas[CD_CLASSIFICACAO])
          )
      
      VAR _Resultado = 
          COUNTROWS(
              FILTER(
                  _TabelaCidadeEspecialidade,
                  NOT ISBLANK([QtVagas]) &&
                  [QtVagas] <= [Valor Param_Intervalo_Vagas]
              )
          )
      
      RETURN
          IF(ISBLANK(_Resultado), 0, _Resultado)
