Tabela: _MEDIDAS
  Colunas:
    - Column [int64]
  Medidas DAX (Regras de Negócio):
    - 01_SUM_COMPRA:
      
      VAR _UltimoSnapshot =
          CALCULATE(
              MAX(BASE_FATO_NECESSIDADE_COMPRA[DT_SNAPSHOT]),
              REMOVEFILTERS(BASE_FATO_NECESSIDADE_COMPRA[DT_SNAPSHOT])
          )
      RETURN
      CALCULATE(
          SUMX(
              VALUES(BASE_FATO_NECESSIDADE_COMPRA[CHAVE]),
              CALCULATE(
                  MAX(BASE_FATO_NECESSIDADE_COMPRA[COMPRA])
              )
          ),
          BASE_FATO_NECESSIDADE_COMPRA[DT_SNAPSHOT] = _UltimoSnapshot
      )
    - 02_SUM_DISP:
      
      VAR _UltimoSnapshot =
          CALCULATE(
              MAX(BASE_FATO_NECESSIDADE_COMPRA[DT_SNAPSHOT]),
              REMOVEFILTERS(BASE_FATO_NECESSIDADE_COMPRA[DT_SNAPSHOT])
          )
      RETURN
      CALCULATE(
          SUMX(
              VALUES(BASE_FATO_NECESSIDADE_COMPRA[CHAVE]),
              CALCULATE(MAX(BASE_FATO_NECESSIDADE_COMPRA[DISPONIBILIZADAS]))
          ),
          BASE_FATO_NECESSIDADE_COMPRA[DT_SNAPSHOT] = _UltimoSnapshot
      )
    - 03_SUM_MARCADAS:
      
      
      VAR _UltimoSnapshot =
          CALCULATE(
              MAX(BASE_FATO_NECESSIDADE_COMPRA[DT_SNAPSHOT]),
              REMOVEFILTERS(BASE_FATO_NECESSIDADE_COMPRA[DT_SNAPSHOT])
          )
      
      RETURN
      CALCULATE(
          SUMX(
              SUMMARIZE(
                  BASE_FATO_NECESSIDADE_COMPRA,
                  BASE_FATO_NECESSIDADE_COMPRA[CHAVE],
                  "MarcadasFoto",
                      MAX(BASE_FATO_NECESSIDADE_COMPRA[MARCADAS])
              ),
              [MarcadasFoto]
          ),
          BASE_FATO_NECESSIDADE_COMPRA[DT_SNAPSHOT] = _UltimoSnapshot
      )
    - 04_(%)ENTREGA:
      
      DIVIDE([02_SUM_DISP], [01_SUM_COMPRA], 0)
    - 05_SALDO_COMPRA:
      '_MEDIDAS'[01_SUM_COMPRA] - '_MEDIDAS'[02_SUM_DISP] 
    - 06_(%)_MARC_VS_DISP:
      DIVIDE('_MEDIDAS'[03_SUM_MARCADAS],'_MEDIDAS'[02_SUM_DISP],0)
    - 0_MSG_BOAS_VINDAS:
      
      VAR vUsuario = LEFT(USERPRINCIPALNAME(), FIND("@", USERPRINCIPALNAME()) - 1)
      VAR vHorarioAtual = NOW() - TIME(3, 0, 0) // Subtrai 3 horas para ajustar ao fuso horário de Brasília (UTC -3)
      VAR vSaudacao = 
          SWITCH(
              TRUE(),
              HOUR(vHorarioAtual) < 12, ", bom dia. Seja bem-vindo(a)!",
              HOUR(vHorarioAtual) < 18, ", boa tarde. Seja bem-vindo(a)!",
              HOUR(vHorarioAtual) <= 23, ", boa noite. Seja bem-vindo(a)!"
          )
      RETURN
      "Olá, " & vUsuario & vSaudacao
    - 07_VOL_ESPECIALIDADE_PREV_COMPRA:
      
      VAR _UltimoSnapshot =
          CALCULATE(
              MAX(BASE_FATO_NECESSIDADE_COMPRA[DT_SNAPSHOT]),
              REMOVEFILTERS(BASE_FATO_NECESSIDADE_COMPRA[DT_SNAPSHOT])
          )
      RETURN
      CALCULATE(
          COUNTROWS(BASE_FATO_NECESSIDADE_COMPRA),
          FILTER(
              BASE_FATO_NECESSIDADE_COMPRA,
              VAR EspecialidadeAtual =
                  BASE_FATO_NECESSIDADE_COMPRA[CD_ESPECIALIDADE]
      
              VAR LocalidadeAtual =
                  BASE_FATO_NECESSIDADE_COMPRA[CD_LOCALIDADE_CIDADE]
      
              VAR FilialAtual =
                  BASE_FATO_NECESSIDADE_COMPRA[CD_FILIAL]
      
              VAR MesAtual =
                  RELATED(DIM_CALENDARIO[MES])
      
              VAR EhMistaDim =
                  CALCULATE(
                      COUNTROWS(DIM_MISTAS),
                      FILTER(
                          ALL(DIM_MISTAS),
                          DIM_MISTAS[CD_ESPECIALIDADE] = EspecialidadeAtual
                              &&
                          DIM_MISTAS[CD_LOCALIDADE] = LocalidadeAtual
                              &&
                          DIM_MISTAS[COD_FILIAL] = FilialAtual
                              &&
                          DIM_MISTAS[MES_FORMATADO] = MesAtual
                      )
                  )
      
              RETURN
                  BASE_FATO_NECESSIDADE_COMPRA[COMPRA] > 0
                      &&
                  BASE_FATO_NECESSIDADE_COMPRA[DT_SNAPSHOT] = _UltimoSnapshot
                      &&
                  UPPER(TRIM(BASE_FATO_NECESSIDADE_COMPRA[GRUPO_ASSERTIVIDADE_MODELO])) <> "MISTAS"
                      &&
                  EhMistaDim = 0
          )
      )
    - 08_VOL. ESPECIALIDADE COM ENTREGA DA NECESSIDADE >90%:
      
      VAR _UltimoSnapshot =
          CALCULATE(
              MAX(BASE_FATO_NECESSIDADE_COMPRA[DT_SNAPSHOT]),
              REMOVEFILTERS(BASE_FATO_NECESSIDADE_COMPRA[DT_SNAPSHOT])
          )
      RETURN
      CALCULATE(
          COUNTROWS(BASE_FATO_NECESSIDADE_COMPRA),
          FILTER(
              BASE_FATO_NECESSIDADE_COMPRA,
              VAR EspecialidadeAtual =
                  BASE_FATO_NECESSIDADE_COMPRA[CD_ESPECIALIDADE]
      
              VAR LocalidadeAtual =
                  BASE_FATO_NECESSIDADE_COMPRA[CD_LOCALIDADE_CIDADE]
      
              VAR FilialAtual =
                  BASE_FATO_NECESSIDADE_COMPRA[CD_FILIAL]
      
              VAR MesAtual =
                  RELATED(DIM_CALENDARIO[MES])
      
              VAR EhMistaDim =
                  CALCULATE(
                      COUNTROWS(DIM_MISTAS),
                      FILTER(
                          ALL(DIM_MISTAS),
                          DIM_MISTAS[CD_ESPECIALIDADE] = EspecialidadeAtual
                              &&
                          DIM_MISTAS[CD_LOCALIDADE] = LocalidadeAtual
                              &&
                          DIM_MISTAS[COD_FILIAL] = FilialAtual
                              &&
                          DIM_MISTAS[MES_FORMATADO] = MesAtual
                      )
                  )
      
              RETURN
                  BASE_FATO_NECESSIDADE_COMPRA[DT_SNAPSHOT] = _UltimoSnapshot
                      &&
                  BASE_FATO_NECESSIDADE_COMPRA[COMPRA] > 0
                      &&
                  UPPER(TRIM(BASE_FATO_NECESSIDADE_COMPRA[GRUPO_ASSERTIVIDADE_MODELO])) <> "MISTAS"
                      &&
                  EhMistaDim = 0
                      &&
                  DIVIDE(
                      BASE_FATO_NECESSIDADE_COMPRA[DISPONIBILIZADAS],
                      BASE_FATO_NECESSIDADE_COMPRA[COMPRA],
                      0
                  ) >= 0.90
          )
      )
    - 09_%_VOL_ESPECIALIDADE_ACIMA_90%:
      
      DIVIDE(
          [08_VOL. ESPECIALIDADE COM ENTREGA DA NECESSIDADE >90%],
          [07_VOL_ESPECIALIDADE_PREV_COMPRA],
          0
      )
    - 10_STATUS_ACIMA_90%:
      
      VAR TabelaResumo =
          SUMMARIZE(
              BASE_FATO_NECESSIDADE_COMPRA,
              DIM_CIDADE[SUB_REGIAO],
              DIM_CIDADE[EMPRESA],
              DIM_CIDADE[NM_CIDADE],
              DIM_NECESSIDADE_COMPRA[DIRETOR],
              DIM_NECESSIDADE_COMPRA[DIRETORIA],
              BASE_FATO_NECESSIDADE_COMPRA[CD_ESPECIALIDADE],
              "Compra", SUM(BASE_FATO_NECESSIDADE_COMPRA[COMPRA]),
              "Disponibilizadas", SUM(BASE_FATO_NECESSIDADE_COMPRA[DISPONIBILIZADAS])
          )
      
      VAR LinhasFiltradas =
          FILTER(
              TabelaResumo,
              DIVIDE([Disponibilizadas], [Compra], 0) >= 0.9
          )
      
      RETURN
      IF(
          COUNTROWS(LinhasFiltradas) > 0,
          "ENTREGUE ✅",
          "NÃO ENTREGUE ❌"
      )
      
    - 11_FILTRO_DADOS_TABELA:
      
      IF(
          CALCULATE(
              COUNTROWS(BASE_FATO_NECESSIDADE_COMPRA)
          ) > 0,
          1,
          BLANK()
      )
      
    - 12_STATUS_ACIMA_90%:
      
      VAR GrupoAtual =
          SELECTEDVALUE(BASE_FATO_NECESSIDADE_COMPRA[GRUPO_ASSERTIVIDADE_MODELO])
      
      VAR EspecialidadeAtual =
          SELECTEDVALUE(BASE_FATO_NECESSIDADE_COMPRA[CD_ESPECIALIDADE])
      
      VAR LocalidadeAtual =
          SELECTEDVALUE(BASE_FATO_NECESSIDADE_COMPRA[CD_LOCALIDADE_CIDADE])
      
      VAR FilialAtual =
          SELECTEDVALUE(BASE_FATO_NECESSIDADE_COMPRA[CD_FILIAL])
      
      VAR MesAtual =
          SELECTEDVALUE(DIM_CALENDARIO[MES])
      
      VAR EhMistaDim =
          CALCULATE(
              COUNTROWS(DIM_MISTAS),
              FILTER(
                  ALL(DIM_MISTAS),
                  DIM_MISTAS[CD_ESPECIALIDADE] = EspecialidadeAtual
                      &&
                  DIM_MISTAS[CD_LOCALIDADE] = LocalidadeAtual
                      &&
                  DIM_MISTAS[COD_FILIAL] = FilialAtual
                      &&
                  DIM_MISTAS[MES_FORMATADO] = MesAtual
              )
          )
      
      VAR Disponibilizada =
          '_MEDIDAS'[02_SUM_DISP]
      
      VAR Perc_Entrega =
          DIVIDE('_MEDIDAS'[02_SUM_DISP], '_MEDIDAS'[01_SUM_COMPRA], 0)
      
      VAR VALOR_FINAL =
          IF(
              GrupoAtual = "MISTAS"
                  || EhMistaDim > 0,
              "MISTA 🔀",
              IF(
                  Disponibilizada = 0,
                  "❌ NÃO ENTREGUE",
                  IF(
                      Perc_Entrega = BLANK(),
                      BLANK(),
                      IF(
                          Perc_Entrega >= 0.9,
                          "✅ ENTREGUE",
                          "❌ NÃO ENTREGUE"
                      )
                  )
              )
          )
      
      RETURN
          VALOR_FINAL
    - 13_STATUS_VOL_ACIMA_90%:
      
      VAR Perc_Entrega = [09_%_VOL_ESPECIALIDADE_ACIMA_90%]
      RETURN
      IF(
          Perc_Entrega >= 0.9,
          "ENTREGUE ✅",

      )

                  "PRAZO", LOOKUPVALUE(

                  )

    - 19_ENTREGA_15_MES_SEGUINTE:

    - HTML_Metas_Tele:
      
      
      VAR _DataSelecionada = MAX('DIM_CALENDARIO'[Date]) 
      VAR _MesFiltro = MONTH(_DataSelecionada)
      VAR _AnoFiltro = YEAR(_DataSelecionada)
      
      VAR _DataSeg = DATE(_AnoFiltro, _MesFiltro, 1)
      VAR _DataRef = EDATE(_DataSeg, -1)
      
      VAR _NomeMesRef = UPPER(FORMAT(_DataRef, "MMM"))
      VAR _NomeMesSeg = UPPER(FORMAT(_DataSeg, "MMM"))
      
      VAR _D15A = DATE(YEAR(_DataRef), MONTH(_DataRef), 15)
      VAR _D20A = DATE(YEAR(_DataRef), MONTH(_DataRef), 20)
      VAR _D01S = DATE(YEAR(_DataSeg), MONTH(_DataSeg), 1)
      VAR _D15S = DATE(YEAR(_DataSeg), MONTH(_DataSeg), 15)
      VAR _D01ProxMes = EDATE(_DataSeg, 1)
      
      VAR _Hoje = TODAY()
      
      VAR _Corte15A = IF(_Hoje >= _D15A, _D15A, _Hoje)
      VAR _Corte20A = IF(_Hoje >= _D20A, _D20A, _Hoje)
      VAR _Corte01S = IF(_Hoje >= _D01S, _D01S, _Hoje)
      VAR _Corte15S = IF(_Hoje >= _D15S, _D15S, _Hoje)
      VAR _CorteFechamento = IF(_Hoje >= _D01ProxMes, _D01ProxMes, _Hoje)
      
      VAR _UltimoSnapshotBase =
          CALCULATE(
              MAX(BASE_FATO_NECESSIDADE_COMPRA[DT_SNAPSHOT]),
              REMOVEFILTERS('DIM_CALENDARIO')
          )
      
      VAR _TabelaBase =
          CALCULATETABLE(
              SUMMARIZE(
                  BASE_FATO_NECESSIDADE_COMPRA,
                  BASE_FATO_NECESSIDADE_COMPRA[CD_ESPECIALIDADE]
              ),
              KEEPFILTERS(BASE_FATO_NECESSIDADE_COMPRA[TIPO_COMPRA] = "COMPRA TELEMEDICINA"),
              ALL('DIM_CALENDARIO'),
              BASE_FATO_NECESSIDADE_COMPRA[MES] = _DataSeg
          )
      
      VAR _TabelaOrdenada =
          ADDCOLUMNS(
              _TabelaBase,
              "@AdFinal",
                  VAR _IdOrd = BASE_FATO_NECESSIDADE_COMPRA[CD_ESPECIALIDADE]
      
                  VAR _SnapCompraOrd =
                      CALCULATE(
                          MAX(BASE_FATO_NECESSIDADE_COMPRA[DT_SNAPSHOT]),
                          REMOVEFILTERS('DIM_CALENDARIO'),
                          BASE_FATO_NECESSIDADE_COMPRA[TIPO_COMPRA] = "COMPRA TELEMEDICINA",
                          BASE_FATO_NECESSIDADE_COMPRA[MES] = _DataSeg,
                          BASE_FATO_NECESSIDADE_COMPRA[CD_ESPECIALIDADE] = _IdOrd
                      )
      
                  VAR _CompraOrd =
                      CALCULATE(
                          SUMX(
                              SUMMARIZE(
                                  BASE_FATO_NECESSIDADE_COMPRA,
                                  BASE_FATO_NECESSIDADE_COMPRA[CHAVE],
                                  "CompraFoto", MAX(BASE_FATO_NECESSIDADE_COMPRA[COMPRA])
                              ),
                              [CompraFoto]
                          ),
                          REMOVEFILTERS('DIM_CALENDARIO'),
                          BASE_FATO_NECESSIDADE_COMPRA[TIPO_COMPRA] = "COMPRA TELEMEDICINA",
                          BASE_FATO_NECESSIDADE_COMPRA[MES] = _DataSeg,
                          BASE_FATO_NECESSIDADE_COMPRA[CD_ESPECIALIDADE] = _IdOrd,
                          BASE_FATO_NECESSIDADE_COMPRA[DT_SNAPSHOT] = _SnapCompraOrd
                      )
      
                  VAR _EntregaOrd =
                      CALCULATE(
                          SUM(BASE_FATO_NECESSIDADE_COMPRA[DISPONIBILIZADAS]),
                          REMOVEFILTERS('DIM_CALENDARIO'),
                          BASE_FATO_NECESSIDADE_COMPRA[TIPO_COMPRA] = "COMPRA TELEMEDICINA",
                          BASE_FATO_NECESSIDADE_COMPRA[MES] = _DataSeg,
                          BASE_FATO_NECESSIDADE_COMPRA[CD_ESPECIALIDADE] = _IdOrd,
                          BASE_FATO_NECESSIDADE_COMPRA[DT_SNAPSHOT] = _SnapCompraOrd
                      )
      
                  RETURN
                      DIVIDE(_EntregaOrd, _CompraOrd, 0)
          )
      
      VAR _GerarLinhas = 
      CONCATENATEX(
          _TabelaOrdenada,
      
          VAR _Id = BASE_FATO_NECESSIDADE_COMPRA[CD_ESPECIALIDADE]
      
          VAR _Esp =
              LOOKUPVALUE(
                  DIM_ESPECIALIDADE[NM_ESPECIALIDADE],
                  DIM_ESPECIALIDADE[CD_ESPECIALIDADE], _Id
              )
      
          VAR _P =
              LOOKUPVALUE(
                  PRAZO_ANS[QT_DIAS_ANS],
                  PRAZO_ANS[CD_ESPECIALIDADE], _Id
              )
      
          VAR _SnapCompra =
              CALCULATE(
                  MAX(BASE_FATO_NECESSIDADE_COMPRA[DT_SNAPSHOT]),
                  REMOVEFILTERS('DIM_CALENDARIO'),
                  BASE_FATO_NECESSIDADE_COMPRA[TIPO_COMPRA] = "COMPRA TELEMEDICINA",
                  BASE_FATO_NECESSIDADE_COMPRA[MES] = _DataSeg,
                  BASE_FATO_NECESSIDADE_COMPRA[CD_ESPECIALIDADE] = _Id
              )
      
          VAR _C =
              CALCULATE(
                  SUMX(
                      SUMMARIZE(
                          BASE_FATO_NECESSIDADE_COMPRA,
                          BASE_FATO_NECESSIDADE_COMPRA[CHAVE],
                          "CompraFoto", MAX(BASE_FATO_NECESSIDADE_COMPRA[COMPRA])
                      ),
                      [CompraFoto]
                  ),
                  REMOVEFILTERS('DIM_CALENDARIO'),
                  BASE_FATO_NECESSIDADE_COMPRA[TIPO_COMPRA] = "COMPRA TELEMEDICINA",
                  BASE_FATO_NECESSIDADE_COMPRA[MES] = _DataSeg,
                  BASE_FATO_NECESSIDADE_COMPRA[CD_ESPECIALIDADE] = _Id,
                  BASE_FATO_NECESSIDADE_COMPRA[DT_SNAPSHOT] = _SnapCompra
              )
      
          VAR _EntregaMes =
              CALCULATE(
                  SUM(BASE_FATO_NECESSIDADE_COMPRA[DISPONIBILIZADAS]),
                  REMOVEFILTERS('DIM_CALENDARIO'),
                  BASE_FATO_NECESSIDADE_COMPRA[TIPO_COMPRA] = "COMPRA TELEMEDICINA",
                  BASE_FATO_NECESSIDADE_COMPRA[MES] = _DataSeg,
                  BASE_FATO_NECESSIDADE_COMPRA[CD_ESPECIALIDADE] = _Id,
                  BASE_FATO_NECESSIDADE_COMPRA[DT_SNAPSHOT] = _SnapCompra
              )
      
          VAR _SnapFechamento =
              CALCULATE(
                  MAX(BASE_FATO_NECESSIDADE_COMPRA[DT_SNAPSHOT]),
                  REMOVEFILTERS('DIM_CALENDARIO'),
                  BASE_FATO_NECESSIDADE_COMPRA[TIPO_COMPRA] = "COMPRA TELEMEDICINA",
                  BASE_FATO_NECESSIDADE_COMPRA[MES] = _DataSeg,
                  BASE_FATO_NECESSIDADE_COMPRA[CD_ESPECIALIDADE] = _Id,
                  BASE_FATO_NECESSIDADE_COMPRA[DT_SNAPSHOT] <= _CorteFechamento
              )
      
          VAR _FechamentoDisp =
              CALCULATE(
                  SUM(BASE_FATO_NECESSIDADE_COMPRA[DISPONIBILIZADAS]),
                  REMOVEFILTERS('DIM_CALENDARIO'),
                  BASE_FATO_NECESSIDADE_COMPRA[TIPO_COMPRA] = "COMPRA TELEMEDICINA",
