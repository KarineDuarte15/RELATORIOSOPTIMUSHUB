Tabela: _MEDIDAS
  Colunas:
    - Column [int64]
  Medidas DAX (Regras de Negócio):
    - 01.1_VAGAS_DIA_ATUAL_CONSULTAS:
      
      
      VAR _vUltimaData = MAX(DIM_CALENDARIO[Date])
      
      RETURN
      CALCULATE(
          '_MEDIDAS'[01_QTD_CONSULTAS],
          ALL(DIM_CALENDARIO[Date]),
          DIM_CALENDARIO[Date] = _vUltimaData
      )
    - 01_QTD_CONSULTAS:
      SUM('FATO_ALERTA_SUB'[QTE_CONSULTAS]) 
    - 02_QTD_ENCAIXES:
      SUM('FATO_ALERTA_SUB'[QTE_ENCAIXES])
    - 03_QTD_CONS_+_ENCAIXE:
      SUM('FATO_ALERTA_SUB'[CONS+ENC])
    - 02.1_VAGAS_DIA_ATUAL_ENCAIXES:
      
      
      VAR _vUltimaData = MAX(DIM_CALENDARIO[Date])
      
      RETURN
      CALCULATE(
          _MEDIDAS[02_QTD_ENCAIXES],
          ALL(DIM_CALENDARIO[Date]),
          DIM_CALENDARIO[Date] = _vUltimaData
      )
      
    - 03.1_VAGAS_DIA_ATUAL_CONS_+_ENC:
      
      
      VAR _vUltimaData = MAX(DIM_CALENDARIO[Date])
      
      RETURN
      CALCULATE(
          _MEDIDAS[03_QTD_CONS_+_ENCAIXE],
          ALL(DIM_CALENDARIO[Date]),
          DIM_CALENDARIO[Date] = _vUltimaData
      )
      
    - 04_VAGAS_MANHA_APENAS_CONSULTAS:
      
      
      CALCULATE(
          '_MEDIDAS'[01_QTD_CONSULTAS],
          FILTER(
              'FATO_ALERTA_SUB',
              'FATO_ALERTA_SUB'[TURNO] = "M" &&
              'FATO_ALERTA_SUB'[HORA] = MIN('FATO_ALERTA_SUB'[HORA])
          
          )
      )
      
    - 04.1_VAGAS_MANHA_COM_ENCAIXES:
      
      CALCULATE(
          '_MEDIDAS'[03_QTD_CONS_+_ENCAIXE],
          FILTER(
              'FATO_ALERTA_SUB',
              'FATO_ALERTA_SUB'[TURNO] = "M" &&
              'FATO_ALERTA_SUB'[HORA] = MIN('FATO_ALERTA_SUB'[HORA])
          
          )
      )
      
    - 05_VAGAS_TARDE_APENAS_CONSULTAS:
      
      CALCULATE(
          '_MEDIDAS'[01_QTD_CONSULTAS],
          FILTER(
              'FATO_ALERTA_SUB',
              'FATO_ALERTA_SUB'[TURNO] = "T" &&
              'FATO_ALERTA_SUB'[HORA] = MAX('FATO_ALERTA_SUB'[HORA])
          
          )
      )
    - 05.1_VAGAS_TARDE_COM_ENCAIXES:
      
      
       CALCULATE(
          '_MEDIDAS'[03_QTD_CONS_+_ENCAIXE],
          FILTER(
              'FATO_ALERTA_SUB',
              'FATO_ALERTA_SUB'[TURNO] = "T" &&
              'FATO_ALERTA_SUB'[HORA] = MAX('FATO_ALERTA_SUB'[HORA])
          
          )
      )
      
    - 01_QTD_DIAS_ZERADOS:
      
      VAR Limite = [Valor VagaDiaZerado]
      RETURN
      IF (
          MINX ( VALUES ( 'FATO_DIAS_ZERADOS'[NM_CIDADE] ), [01_MAX_DIAS_ZERADOS] ) = 0,
          0,
          MAXX ( VALUES ( 'FATO_DIAS_ZERADOS'[NM_CIDADE] ), [01_MAX_DIAS_ZERADOS] )
      )
      
    - 01_MAX_DIAS_ZERADOS:
      MAX(FATO_DIAS_ZERADOS[QT_DIAS_ZERADOS])
