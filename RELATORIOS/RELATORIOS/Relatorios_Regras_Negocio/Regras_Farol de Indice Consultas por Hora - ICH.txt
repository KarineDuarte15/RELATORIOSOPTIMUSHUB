Tabela: #MEDIDAS
  Medidas DAX (Regras de Negócio):
    - Qtde Consultas:
      SUM( fProdutividade[Qtd. Consultas] )
    - Qtde Retorno:
      SUM( fProdutividade[Qtd. Retorno] )
    - Qtde Horas:
      SUM ( fProdutividade[Qtd. Horas] )
    - Produtividade:
      DIVIDE( [Total Atendimentos] , [Qtde Horas] ) 
    - Produtividade Mes Atual:
      
      VAR MaxData =
          CALCULATE(
              MAX(dCalendario[Date]),
              ALL(dCalendario)
          )
      VAR MesAnoMax =
          CALCULATE(
              MAX(dCalendario[Mes_Ano]),
              FILTER(ALL(dCalendario), dCalendario[Date] = MaxData)
          )
      VAR AtendMes =
          CALCULATE(
              SUM(fProdutividade[Atendimento Geral]),
              FILTER(ALL(dCalendario), dCalendario[Mes_Ano] = MesAnoMax)
          )
      VAR HorasMes =
          CALCULATE(
              SUM(fProdutividade[Qtd. Horas]),
              FILTER(ALL(dCalendario), dCalendario[Mes_Ano] = MesAnoMax)
          )
      RETURN
      DIVIDE(AtendMes, HorasMes)
      
    - Qtde Consultas Mes Atual:
      
      CALCULATE(
          SUM(fProdutividade[Qtd. Consultas]),
          FILTER(
              dCalendario,
              dCalendario[Ano] = YEAR(TODAY()) && dCalendario[Mes] = MONTH(TODAY())
          )
      )
    - Qtde Horas Mes Atual:
      
      VAR MesAnoAtual =
          FORMAT ( TODAY(), "mmm/yy" )
      VAR MesAnoAnterior =
          FORMAT ( EOMONTH ( TODAY(), -1 ), "mmm/yy" )
      VAR HorasMesAtual =
          CALCULATE (
              SUM ( fProdutividade[Qtd. Horas] ),
              FILTER ( dCalendario, dCalendario[Mes_Ano] = MesAnoAtual )
          )
      VAR HorasMesAnterior =
          CALCULATE (
              SUM ( fProdutividade[Qtd. Horas] ),
              FILTER ( dCalendario, dCalendario[Mes_Ano] = MesAnoAnterior )
          )
      RETURN
      IF ( NOT ISBLANK ( HorasMesAtual ) && HorasMesAtual > 0, HorasMesAtual, HorasMesAnterior )
      
    - Qtde Retorno Mes Atual:
      
      CALCULATE(
          SUM(fProdutividade[Qtd. Retorno]),
          FILTER(
              dCalendario,
              dCalendario[Ano] = YEAR(TODAY()) && dCalendario[Mes] = MONTH(TODAY())
          )
      )
    - Total Atendimentos:

    - Total Atendimentos Mes Atual:

    - Qtde Consultas MoM:
      
      CALCULATE(
          SUM(fProdutividade[Qtd. Consultas]),
          FILTER(
              dCalendario,
              dCalendario[Ano] = YEAR(EOMONTH(TODAY(), -1)) &&

      )
    - Qtde Consultas YoY:
      
      CALCULATE(

    - Qtde Horas MoM:
      
      VAR _MaxData =
          MAX ( dCalendario[Date] )
      VAR _AnoAnterior =
          YEAR ( EOMONTH ( _MaxData, -1 ) )
      VAR _MesAnterior =
          MONTH ( EOMONTH ( _MaxData, -1 ) )
      RETURN
      CALCULATE (
          SUM ( fProdutividade[Qtd. Horas] ),
          FILTER (
              dCalendario,
              dCalendario[Ano] = _AnoAnterior
                  && dCalendario[Mes] = _MesAnterior
          )
      )
      
      
      
    - Qtde Horas YoY:
      
      VAR _MaxData =
          MAX ( dCalendario[Date] )
      VAR _AnoRef =
          YEAR ( _MaxData )
      VAR _MesRef =
          MONTH ( _MaxData )
      RETURN
      CALCULATE (
          SUM ( fProdutividade[Qtd. Horas] ),
          FILTER (
              dCalendario,
              dCalendario[Ano] = _AnoRef - 1
                  && dCalendario[Mes] = _MesRef
          )
      )

      VAR _HorasEmpresaMesAnterior =
          CALCULATE(
              SUM(fProdutividade[Qtd. Horas]),

                  dCalendario,

          CALCULATE(
