--- TABELAS, COLUNAS E MEDIDAS (DAX) ---

Tabela: FatoAgendas
  Colunas:
    - EMPRESA [string]
    - FILIAL [string]
    - CD_LOCAL [int64]
    - UNIDADE [string]
    - CD_MEDICO [int64]
    - MÉDICO [string]
    - ESPECIALIDADE [string]
    - STATUS DA NÃO ABERTURA DE AGENDA [string]
    - Nome da Origem [string]
    - Mes_Num [string]
    - Ano [string]
    - Data_Referencia [dateTime]
    - Classificacao Status [string] (Calculada)
    - CD_LOCALIDADE [string]
    - ChaveLocalEsp [string]
    - CD_CIDADE_IBGE [int64]
    - CD_CLASSIFICACAO [string]
    - Status Padronizado [string] (Calculada)
    - CD_ESPECIALIDADE [string]
    - REGIONAL [string]
    - RESPONSAVEL [string]
    - CIDADE [string]
    - UF [string]
    - ChaveLocalEsp_Display [string] (Calculada)
    - ChaveLocalEsp_Final [string] (Calculada)
    - ChaveCidade [string] (Calculada)
    - REGIONAL_FILTER [string] (Calculada)
    - EMPRESA_FILTER [string] (Calculada)
    - POLO_NOVO [string]
    - ChaveUnidade [string] (Calculada)
----------------------------------------
Tabela: CALENDARIO
  Colunas:
    - Data [dateTime]
    - Ano [int64]
    - Mes_Num [int64]
    - Mes_Nome [string]
    - Dia [int64]
    - MesAno_Ordem [int64] (Calculada)
    - Mes_Ano [string] (Calculada)
----------------------------------------
Tabela: Medidas
  Colunas:
    - Column [int64]
  Medidas DAX (Regras de Negócio):
    - Qtd Agendas Positivas:
      
      CALCULATE(
          COUNTROWS(FatoAgendas),
          FatoAgendas[Classificacao Status] = "Positivo"
      )
    - % Entrega:
      
      DIVIDE (
          [Qtd Agendas Positivas],
          [Qtd Agendas Consideradas]
      )
      
      
    - Tem Dados:
      
      IF (
          [Qtd Agendas Consideradas] > 0,
          1,
          0
      )
      
    - Qtd Agendas Consideradas:
      
      CALCULATE(
          COUNTROWS(FatoAgendas),
          FatoAgendas[Classificacao Status] <> "Nao Considerar"
      )
      
      
      
      
      
    - Qtd Agendas Negativas:
      
      CALCULATE(
          COUNTROWS(FatoAgendas),
          FatoAgendas[Classificacao Status] = "Negativo"
      )
      
    - Meta % Entrega:
      
      VAR CorteAtivo = SELECTEDVALUE(dCorte[Corte])
      RETURN
          SWITCH(
              TRUE(),
              CorteAtivo = "1DU",     0.90,
              CorteAtivo = "5DU",     0.95,
              CorteAtivo = "10DU",    1.00,
              1.00
          )
      
    - % Acumulado Não Entregues:
      
      VAR TotalNegativas =
          CALCULATE (
              [Qtd Agendas Negativas],
              ALL ( dCidade[EMPRESA] )
          )
      
      VAR ValorAtual =
          [Qtd Agendas Negativas]
      
      VAR Acumulado =
          CALCULATE (
              [Qtd Agendas Negativas],
              FILTER (
                  ALL ( dCidade[EMPRESA] ),
                  [Qtd Agendas Negativas] >= ValorAtual
              )
          )
      
      RETURN
      DIVIDE ( Acumulado, TotalNegativas )
      
      
    - Tipo Visual Status:
      
      IF (
          SELECTEDVALUE ( FatoAgendas[Classificacao Status] ) = "Positivo",
          "Positivo",
          "Negativo"
      )
      
    - Qtd Agendas Consideradas Mês Anterior:
      
      CALCULATE (
          [Qtd Agendas Consideradas],
          PREVIOUSMONTH ( CALENDARIO[Data] )
      )
      
    - Variação % vs Mês Anterior:
      
      VAR Atual = [Qtd Agendas Consideradas]
      VAR Anterior = [Qtd Agendas Consideradas Mês Anterior]
      RETURN
      DIVIDE ( Atual - Anterior, Anterior )
      
    - Variação Formatada:
      
      VAR VarPerc = [Variação % vs Mês Anterior]
      RETURN
      IF (
          ISBLANK ( VarPerc ),
          BLANK(),
          IF (
              VarPerc > 0,
              "▲ " & FORMAT ( VarPerc, "0.0%" ),
              "▼ " & FORMAT ( VarPerc, "0.0%" )
          )
      )
      
    - Cor Variação:
      
      IF (
          [Variação % vs Mês Anterior] >= 0,
          "#1A8F3C",   -- verde
          "#D64545"    -- vermelho
      )
      
    - Qtd Entregues Mês Anterior:
      
      CALCULATE (
          [Qtd Agendas Positivas],
          PREVIOUSMONTH ( CALENDARIO[Data] )
      )
      
    - Variação Entregues %:
      
      VAR Atual = [Qtd Agendas Positivas]
      VAR Anterior = [Qtd Entregues Mês Anterior]
      RETURN
      DIVIDE ( Atual - Anterior, Anterior )
      
    - Variação Entregues Formatada:
      
      VAR VarPerc = [Variação Entregues %]
      RETURN
      IF (
          ISBLANK ( VarPerc ),
          BLANK(),
          IF (
              VarPerc > 0,
              "▲ " & FORMAT ( VarPerc, "0.0%" ),
              "▼ " & FORMAT ( VarPerc, "0.0%" )
          )
      )
      
    - Cor Variação Entregues:
      
      VAR VarPerc = [Variação Entregues %]
      RETURN
      SWITCH(
          TRUE(),
          ISBLANK(VarPerc), "#6C757D",
          VarPerc > 0, "#1A8F3C",
          VarPerc < 0, "#D64545",
          "#6C757D"
      )
      
      
      
    - Qtd Não Entregues Mês Anterior:
      
      CALCULATE (
          [Qtd Agendas Negativas],
          PREVIOUSMONTH ( CALENDARIO[Data] )
      )
      
    - Variação Não Entregues %:
      
      VAR Atual = [Qtd Agendas Negativas]
      VAR Anterior = [Qtd Não Entregues Mês Anterior]
      RETURN
      DIVIDE ( Atual - Anterior, Anterior )
      
    - Variação Não Entregues Formatada:
      
      VAR VarPerc = [Variação Não Entregues %]
      RETURN
      IF (
          ISBLANK ( VarPerc ),
          BLANK(),
          IF (
              VarPerc > 0,
              "▲ " & FORMAT ( VarPerc, "0.0%" ),
              "▼ " & FORMAT ( VarPerc, "0.0%" )
          )
      )
      
    - Cor Variação Não Entregues:
      
      VAR VarPerc = [Variação Não Entregues %]
      RETURN
      SWITCH(
          TRUE(),
          ISBLANK(VarPerc), "#6C757D",
          VarPerc > 0, "#D64545",   -- aumentou negativas = ruim
          VarPerc < 0, "#1A8F3C",   -- diminuiu negativas = bom
          "#6C757D"
      )
      
    - Cor % Entrega:
      
      IF(
          [% Entrega] >= 0.9,
          "#3A9D7A",
          "#F39C12"
      )
      
    - Gap para Meta (p.p):
      
      [Meta % Entrega] - [% Entrega]
      
    - Falta para Meta:
      
      VAR Atual = [% Entrega]
      VAR Meta = [Meta % Entrega]
      RETURN
      IF(
          Atual >= Meta,
          0,
          Meta - Atual
      )
      
    - Atingido da Meta:
      
      MIN([% Entrega], [Meta % Entrega])
      
    - Nome Mês Anterior:
      
      VAR MesAnterior = 
          CALCULATE(
              MAX(CALENDARIO[Data]),
              PREVIOUSMONTH(CALENDARIO[Data])
          )
      RETURN
      IF(
          ISBLANK(MesAnterior),
          "mês anterior",
          FORMAT(MesAnterior, "MMMM/YY", "pt-BR")
      )
    - Frase Meta Entrega:
      
      VAR Atual = [% Entrega]
      VAR Meta = [Meta % Entrega]
      VAR Gap = Meta - Atual
      RETURN
          SWITCH(
              TRUE(),
              ISBLANK(Atual) || Atual = 0, "Dados não disponíveis",
              Atual >= Meta, "Meta de " & FORMAT(Meta, "0%") & " superada em " & FORMAT(ABS(Gap), "0.0%"),
              Atual < Meta, "Faltam " & FORMAT(ABS(Gap), "0.0%") & " para atingir a meta de " & FORMAT(Meta, "0%")
          )
    - Qtd Consideradas Mês Anterior:
      
      CALCULATE(
          [Qtd Agendas Consideradas],
          PREVIOUSMONTH(CALENDARIO[Data])
      )
    - Qtd Entregues Mês Anterior Label:
      
      VAR Valor = [Qtd Entregues Mês Anterior]
      VAR Mes = [Nome Mês Anterior]
      RETURN
      IF(ISBLANK(Valor), BLANK(), FORMAT(Valor, "#,##0") & " (" & Mes & ")")
    - Qtd Não Entregues Mês Anterior Label:
      
      VAR Valor = [Qtd Não Entregues Mês Anterior]
      VAR Mes = [Nome Mês Anterior]
      RETURN
      IF(ISBLANK(Valor), BLANK(),FORMAT(Valor, "#,##0") & " (" & Mes & ")")
    - Qtd Consideradas Mês Anterior Label:
      
      VAR Valor = [Qtd Consideradas Mês Anterior]
      VAR Mes = [Nome Mês Anterior]
      RETURN
      IF(
          ISBLANK(Valor),
          BLANK(),
          FORMAT(Valor, "#,##0") & " (" & Mes & ")"
      )
    - % do Total Negativos:
      
      VAR TotalGeral = CALCULATE(
          [Qtd Agendas Negativas],
          REMOVEFILTERS(FatoAgendas[FILIAL]),
          REMOVEFILTERS(FatoAgendas[UNIDADE])
      )
      VAR TotalFilial = CALCULATE(
          [Qtd Agendas Negativas],
          REMOVEFILTERS(FatoAgendas[UNIDADE])
      )
      RETURN
      IF(
          ISBLANK([Qtd Agendas Negativas]),
          BLANK(),
          IF(
              ISINSCOPE(FatoAgendas[UNIDADE]),
              DIVIDE([Qtd Agendas Negativas], TotalFilial),
              DIVIDE([Qtd Agendas Negativas], TotalGeral)
          )
      )
    - Variação Negativos vs Mês Anterior:
      
      VAR MesAtual = CALCULATE(
          MAX(CALENDARIO[Mes_Num]),
          ALLSELECTED(CALENDARIO)
      )
      VAR AnoAtual = CALCULATE(
          MAX(CALENDARIO[Ano]),
          ALLSELECTED(CALENDARIO)
      )
      VAR Atual = CALCULATE(
          [Qtd Agendas Negativas],
          CALENDARIO[Mes_Num] = MesAtual,
          CALENDARIO[Ano] = AnoAtual
      )
      VAR Anterior = CALCULATE(
          [Qtd Agendas Negativas],
          PREVIOUSMONTH(CALENDARIO[Data]),
          CALENDARIO[Mes_Num] = MesAtual - 1
      )
      RETURN
      IF(
          ISBLANK(Anterior) || ISBLANK(Atual),
          BLANK(),
          DIVIDE(Atual - Anterior, Anterior)
      )
    - Variação Negativos Formatada:
      
      VAR MesAtual = CALCULATE(MAX(CALENDARIO[Mes_Num]), ALLSELECTED())
      VAR AnoAtual = CALCULATE(MAX(CALENDARIO[Ano]), ALLSELECTED())
      VAR MesAnteriorNum = IF(MesAtual = 1, 12, MesAtual - 1)
      VAR AnoAnteriorNum = IF(MesAtual = 1, AnoAtual - 1, AnoAtual)
      VAR NegAtual = CALCULATE(
          [Qtd Agendas Negativas],
          FILTER(ALL(CALENDARIO), CALENDARIO[Mes_Num] = MesAtual && CALENDARIO[Ano] = AnoAtual)
      )
      VAR NegAnterior = CALCULATE(
          [Qtd Agendas Negativas],
          FILTER(ALL(CALENDARIO), CALENDARIO[Mes_Num] = MesAnteriorNum && CALENDARIO[Ano] = AnoAnteriorNum)
      )
      VAR Variacao = DIVIDE(NegAtual - NegAnterior, NegAnterior)
      RETURN
      IF(
          ISBLANK(NegAnterior) || ISBLANK(NegAtual), "—",
          IF(NegAnterior <= 2, "—",
          IF(Variacao > 0, "▲ " & FORMAT(ABS(Variacao), "0%"),
          IF(Variacao < 0, "▼ " & FORMAT(ABS(Variacao), "0%"),
          "► 0%")))
      )
    - Qtd Cobrando Agenda:
      
      VAR Resultado = CALCULATE(
          COUNTROWS(FatoAgendas),
          FatoAgendas[STATUS DA NÃO ABERTURA DE AGENDA] = "COBRANDO AGENDA"
      )
      RETURN IF(ISBLANK(Resultado), "-", Resultado)
    - Qtd Negociando Credenciamento:
      
      VAR Resultado = CALCULATE(
          COUNTROWS(FatoAgendas),
          FatoAgendas[STATUS DA NÃO ABERTURA DE AGENDA] = "NEGOCIANDO COM CREDENCIAMENTO"
      )
      RETURN IF(ISBLANK(Resultado),"-",Resultado)
    - Qtd Médico Pendência:
      
      VAR Resultado = CALCULATE(
          COUNTROWS(FatoAgendas),
          FatoAgendas[STATUS DA NÃO ABERTURA DE AGENDA] = "MÉDICO COM PENDÊNCIA DE PAGAMENTOS"
      )
      RETURN IF(ISBLANK(Resultado),"-",Resultado)
    - Qtd Consideradas Diretor:
      
      CALCULATE(
          [Qtd Agendas Consideradas],
          USERELATIONSHIP(FatoAgendas[ChaveLocalEsp_Final], 'dDiretoria'[ID])
      )
    - Qtd Entregues Diretor:
      
      CALCULATE(
          [Qtd Agendas Positivas],
          USERELATIONSHIP(FatoAgendas[ChaveLocalEsp_Final], 'dDiretoria'[ID])
      )
    - % Entrega Diretor:
      
      DIVIDE(
          [Qtd Entregues Diretor],
          [Qtd Consideradas Diretor]
      )
    - Status Meta Diretor:
      
      IF(
          ISBLANK([% Entrega Diretor]),
          BLANK(),
          IF([% Entrega Diretor] >= 0.90, "✅ Atingiu", "⚠️ Abaixo")
      )
    - % Não Entrega Diretor:
      
      1 - [% Entrega Diretor]
    - Qtd Não Entregues Diretor:
      
      [Qtd Consideradas Diretor] - [Qtd Entregues Diretor]
    - Variação Entrega Diretor_format:
      
      VAR MesAtual = CALCULATE(MAX(CALENDARIO[Mes_Num]), ALLSELECTED(CALENDARIO))
      VAR AnoAtual = CALCULATE(MAX(CALENDARIO[Ano]), ALLSELECTED(CALENDARIO))
      VAR MesAnteriorNum = IF(MesAtual = 1, 12, MesAtual - 1)
      VAR AnoAnteriorNum = IF(MesAtual = 1, AnoAtual - 1, AnoAtual)
      VAR PctAtual = CALCULATE(
          [% Entrega],
          FILTER(ALL(CALENDARIO), CALENDARIO[Mes_Num] = MesAtual && CALENDARIO[Ano] = AnoAtual)
      )
      VAR PctAnterior = CALCULATE(
          [% Entrega],
          FILTER(ALL(CALENDARIO), CALENDARIO[Mes_Num] = MesAnteriorNum && CALENDARIO[Ano] = AnoAnteriorNum)
      )
      VAR Variacao = PctAtual - PctAnterior
      RETURN
          IF(
              ISBLANK(PctAnterior) || ISBLANK(PctAtual), "—",
              IF(Variacao > 0, "▲ " & FORMAT(ABS(Variacao), "0.0%"),
              IF(Variacao < 0, "▼ " & FORMAT(ABS(Variacao), "0.0%"),
              "► 0.0%"))
          )
    - Qtd Médicos com Pendência:
      
      CALCULATE(
          DISTINCTCOUNT(FatoAgendas[MÉDICO]),
          FatoAgendas[Classificacao Status] = "Negativo"
      )
    - Qtd Médicos 100% Entrega:
      
      CALCULATE(
          DISTINCTCOUNT(FatoAgendas[MÉDICO]),
          FatoAgendas[Classificacao Status] = "Positivo"
      )
    - % Entrega Médico:
      
      VAR Positivas = CALCULATE(COUNTROWS(FatoAgendas), FatoAgendas[Classificacao Status] = "Positivo") + 0
      VAR Consideradas = CALCULATE(COUNTROWS(FatoAgendas), FatoAgendas[Classificacao Status] <> "Nao Considerar")
      RETURN
      IF(
          ISBLANK(Consideradas) || Consideradas = 0, BLANK(),
          DIVIDE(Positivas, Consideradas, 0)
      )
    - Status Médico:
      
      VAR Pct = [% Entrega Médico]
      VAR Consideradas = [Qtd Agendas Consideradas]
      RETURN
      IF(
          ISBLANK(Consideradas) || Consideradas = 0, BLANK(),
          IF(Pct >= 0.90, "✅ Ótimo",
          IF(Pct >= 0.51, "⚠️ Atenção",
          "🔴 Crítico"))
      )

      CALCULATE(
          DISTINCTCOUNT(FatoAgendas[CD_MEDICO]),

      )

    - Qtd Médicos Ótimos:
      
      CALCULATE(
          DISTINCTCOUNT(FatoAgendas[CD_MEDICO]),
          FILTER(
              VALUES(FatoAgendas[CD_MEDICO]),
              VAR Positivas = CALCULATE(COUNTROWS(FatoAgendas), FatoAgendas[Classificacao Status] = "Positivo")
              VAR Consideradas = CALCULATE(COUNTROWS(FatoAgendas), FatoAgendas[Classificacao Status] <> "Nao Considerar")
              RETURN Consideradas > 0 && DIVIDE(Positivas, Consideradas) >= 0.90
          )
      )
    - Qtd Médicos Atenção:
      
      CALCULATE(
          DISTINCTCOUNT(FatoAgendas[CD_MEDICO]),
          FILTER(
              VALUES(FatoAgendas[CD_MEDICO]),
              VAR Positivas = CALCULATE(COUNTROWS(FatoAgendas), FatoAgendas[Classificacao Status] = "Positivo")
              VAR Consideradas = CALCULATE(COUNTROWS(FatoAgendas), FatoAgendas[Classificacao Status] <> "Nao Considerar")
              VAR Pct = DIVIDE(Positivas, Consideradas, 0)
              RETURN Consideradas > 0 && Pct > 0.50 && Pct < 0.90
          )
      )
    - Qtd Médicos Críticos:
      
      CALCULATE(
          DISTINCTCOUNT(FatoAgendas[CD_MEDICO]),
          FILTER(
              VALUES(FatoAgendas[CD_MEDICO]),
              VAR Positivas = CALCULATE(COUNTROWS(FatoAgendas), FatoAgendas[Classificacao Status] = "Positivo")
              VAR Consideradas = CALCULATE(COUNTROWS(FatoAgendas), FatoAgendas[Classificacao Status] <> "Nao Considerar")
              VAR Pct = DIVIDE(Positivas, Consideradas)
              RETURN Consideradas > 0 && Pct <= 0.50
          )
      )

      RETURN
      SWITCH(
          TRUE(),
          ISBLANK(PctAnterior) || ISBLANK(PctAtual), "#808080",
          Variacao > 0, "#2E7D32",   -- verde
          Variacao < 0, "#C62828",   -- vermelho
          "#808080"
      )
    - Cor Performance Diretor:
      
      VAR MesAtual = CALCULATE(MAX(FatoAgendas[Mes_Num]), ALL(FatoAgendas))
      VAR AnoAtual = CALCULATE(MAX(FatoAgendas[Ano]), ALL(FatoAgendas))
      
      VAR Entrega =
      DIVIDE(
          CALCULATE(
              [Qtd Entregues Diretor],
              ALL(CALENDARIO),
              CALENDARIO[Mes_Num] = INT(MesAtual),
              CALENDARIO[Ano] = INT(AnoAtual)
          ),
          CALCULATE(
              [Qtd Consideradas Diretor],
              ALL(CALENDARIO),
              CALENDARIO[Mes_Num] = INT(MesAtual),
              CALENDARIO[Ano] = INT(AnoAtual)
          )
      )
      
      RETURN
      SWITCH(
          TRUE(),
          ISBLANK(Entrega), "#808080",
          Entrega >= 0.90, "#2E7D32",   -- verde
          Entrega < 0.90, "#C62828"     -- vermelho
      )
    - Pct Hist Dinamico:
      
      VAR Positivas =
          CALCULATE(
              COUNTROWS(FatoHistorico),
              FatoHistorico[Classificacao Status Hist] = "Positivo"
          )
      VAR Consideradas =
          CALCULATE(
              COUNTROWS(FatoHistorico),
              FatoHistorico[Classificacao Status Hist] <> "Nao Considerar"
          )
      RETURN
          IF(
              ISBLANK(Consideradas) || Consideradas = 0,
              0,
              DIVIDE(Positivas, Consideradas)
          )
    - Hist Positivas:
      
      CALCULATE(
          COUNTROWS(FatoHistorico),
          FatoHistorico[Classificacao Status Hist] = "Positivo"
      )
    - Hist Consideradas:
      
      CALCULATE(
          COUNTROWS(FatoHistorico),
          FatoHistorico[Classificacao Status Hist] <> "Nao Considerar"
      )
    - ##Msg Boas Vindas:
      
      VAR vUsuario = LEFT(USERPRINCIPALNAME(), FIND("@", USERPRINCIPALNAME()) - 1)
      VAR vHorarioAtual = NOW() - TIME(3, 0, 0) // Subtrai 3 horas para ajustar ao fuso horário de Brasília (UTC -3)
      VAR vSaudacao = 
          SWITCH(
              TRUE(),
              HOUR(vHorarioAtual) < 12, ", bom dia. Seja bem-vindo(a)!",
              HOUR(vHorarioAtual) < 18, ", boa tarde. Seja bem-vindo(a)!",
              HOUR(vHorarioAtual) <= 23, ", boa noite. Seja bem-vindo(a)!"
          )
      RETURN
      "Olá, " & vUsuario & vSaudacao
    - Mai/26:
      
      VAR _MesAno = "mai/2026"
      VAR _Resultado = DIVIDE(
          CALCULATE([Qtd Entregues Diretor], ALL(CALENDARIO), CALENDARIO[Mes_Ano] = _MesAno),
          CALCULATE([Qtd Consideradas Diretor], ALL(CALENDARIO), CALENDARIO[Mes_Ano] = _MesAno)
      )
      RETURN
      IF(ISBLANK(_Resultado), "-", FORMAT(_Resultado, "0.0%"))
    - Mes_Filtrado_Label:
      
      VAR AnoSelecionado = SELECTEDVALUE(CALENDARIO[Ano])
      VAR MesSelecionado = SELECTEDVALUE(CALENDARIO[Mes_Nome])
      RETURN
          IF(
              NOT ISBLANK(AnoSelecionado) && NOT ISBLANK(MesSelecionado),
              "| " & MesSelecionado & "/" & FORMAT(AnoSelecionado, "0000"),
              "Todos os períodos"
          )
    - Jun/26:
      
      VAR _MesAno = "jun/2026"
      VAR _Resultado = DIVIDE(
          CALCULATE([Qtd Entregues Diretor], ALL(CALENDARIO), CALENDARIO[Mes_Ano] = _MesAno),
          CALCULATE([Qtd Consideradas Diretor], ALL(CALENDARIO), CALENDARIO[Mes_Ano] = _MesAno)
      )
      RETURN
      IF(ISBLANK(_Resultado), "-", FORMAT(_Resultado, "0.0%"))
    - Debug Diferenca:
      
      VAR TotalAtual = [Qtd Agendas Consideradas]
      VAR Total5DU = [Qtd Agendas Consideradas D]
      RETURN
          TotalAtual - Total5DU
    - Debug Mes:
      SELECTEDVALUE(CALENDARIO[Mes_Num])
