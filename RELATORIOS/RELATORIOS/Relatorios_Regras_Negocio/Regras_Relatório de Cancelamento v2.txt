Tabela: #MEDIDAS
  Medidas DAX (Regras de Negócio):
    - Qt Falta Medico:
      SUM(fRemarcacao[QT_FALTA_MED])
    - Qt Falta Paciente:
      SUM(fRemarcacao[QT_FALTA_PAC])
    - Qt Rem Paciente:
      SUM(fRemarcacao[QT_REM_PAC])
    - Qt Rem Adm:
      SUM(fRemarcacao[QT_REM_ADM])
    - Qt Rem Medico:
      SUM(fRemarcacao[QT_REM_MED])
    - Qt Efetivadas:
      SUM(fRemarcacao[QT_EFETIVADO])
    - Qt Canceladas:
      [Qt Falta Medico] + [Qt Falta Paciente] + [Qt Rem Adm] + [Qt Rem Medico]
    - Qt Total Consultas:
      [Qt Efetivadas] + [Qt Canceladas]
    - % Falta Medico:
      DIVIDE( [Qt Falta Medico] , [Qt Total Consultas] ) 
    - % Falta Paciente:
      DIVIDE( [Qt Falta Paciente] , [Qt Total Consultas] ) 
    - % Remarcação Adm:
      DIVIDE( [Qt Rem Adm] , [Qt Total Consultas] ) 
    - % Remarcação Medico:
      DIVIDE( [Qt Rem Medico] , [Qt Total Consultas] + [Qt Rem Medico] ) 
    - % Remarcação Paciente:
      DIVIDE( [Qt Rem Paciente] , [Qt Total Consultas] ) 
    - Total Reagendamento:
      SUM( fRemarcacao[QT_REAGENDADO] )
    - Qt Reagend 0_15:
      SUM( fRemarcacao[QT_REAGEND_0_15_DIAS] )
    - Qt Reagend 15_30:
      SUM( fRemarcacao[QT_REAGEND_15_30_DIAS] )
    - Qt Reagend 30_45:
      SUM( fRemarcacao[QT_REAGEND_30_45_DIAS] )
    - Qt Reagend 45_60:
      SUM( fRemarcacao[QT_REAGEND_45_60_DIAS] )
    - Total Cancelamentos:
      SUM( fRemarcacao[QT_CANCELADO1] )
    - % Reagend 0_15:
      DIVIDE( [Qt Reagend 0_15] , [Total Cancelamentos] )
    - % Reagend 15_30:
      DIVIDE( [Qt Reagend 15_30] , [Total Cancelamentos] )
    - % Reagend 30_45:
      DIVIDE( [Qt Reagend 30_45] , [Total Cancelamentos] )
    - % Reagend 45_60:
      DIVIDE( [Qt Reagend 45_60] , [Total Cancelamentos] )
    - % Reagendamentos:
      DIVIDE( [Total Reagendamento] , [Total Cancelamentos] )
    - Qt Reagend Acima 60:
      SUM( fRemarcacao[QT_REAGEND_ACIMA_60_DIAS] )
    - % Reagend Acima 60:
      DIVIDE( [Qt Reagend Acima 60] , [Total Cancelamentos] )
    - % Reagend até 60 dias:
      
      VAR QtTotal = [Qt Reagend 0_15] + [Qt Reagend 15_30] + [Qt Reagend 30_45] + [Qt Reagend 45_60]
      VAR TotalCancelamento = [Total Cancelamentos]
      RETURN
      DIVIDE(  QtTotal , [Total Cancelamentos] )
    - Qt Registros:
      COUNTROWS( fRemarcacao )
    - Qt Canc. < 3H:
      
          SUM(fRemarcacao[QT_CANC_ATE_3H])
    - Top10Motivos:
      
      VAR Top10Motivos = 
          CALCULATETABLE(
              VALUES(dMotivos[DS_MOTIVO_CANCELAMENTO]),
              TOPN(
                  10,
                  ALL(dMotivos[DS_MOTIVO_CANCELAMENTO]),
                  CALCULATE(SUM(fRemarcacao[QT_CANCELADO1])),
                  DESC
              )
          )
      RETURN
          CALCULATE(
              SUM(fRemarcacao[QT_CANCELADO1]),
              KEEPFILTERS(dMotivos[DS_MOTIVO_CANCELAMENTO] IN Top10Motivos)
          )
      
    - Qt Emp. Conveniada:
      DISTINCTCOUNT(fRemarcacao[CD_EMPRESA_CONVENIADA])
