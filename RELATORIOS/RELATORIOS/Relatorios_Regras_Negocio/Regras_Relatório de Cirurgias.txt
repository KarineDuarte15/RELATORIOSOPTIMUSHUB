  Colunas:

    - CD_ATENDIMENTO [string]
    - CD_PROCEDIMENTO [int64]
    - NR_PROCEDIMENTO [string]
    - DT_INICIO_CIRURGIA [dateTime]
    - NM_PACIENTE [string]
    - DT_NASCIMENTO [dateTime]
    - CD_SALA_CIRURGICA [string]
    - NM_MOTIVO_ATENDIMENTO [string]
    - NM_UNIDADE_ATENDIMENTO [string]
    - FL_FONTE [string]
    - QTD [double]
    - CD_UNIDADE_ATENDIMENTO [string]
    - CANCELADO [string]
    - TIPO [string]
    - CONVENIO [string]
    - NM_MEDICO [string]
    - DS_SALA_CIRURGICA [string]
    - TIPO_1 [string] (Calculada)
    - TIPO_AJUSTADO [string] (Calculada)
    - TIPO_ATO [string]
    - CD_PESSOA [double]
  Medidas DAX (Regras de Negócio):
    - Qtd Cirurgias:
      SUM(Cirurgias[QTD])
    - Total Cirurgias:
      
          var mes = SELECTEDVALUE(_CALENDARIO[Mes/Ano])
          var unidade = SELECTEDVALUE(Dim_Unidades[TIPO])
          var convenio = SELECTEDVALUE(Dim_Convenio[TIPO CONVENIO])
      
          RETURN
      
          CALCULATE(
              SUM(Cirurgias[QTD]),
              Cirurgias[CANCELADO] <> "S",
              FILTER(Dim_Convenio, Dim_Convenio[TIPO CONVENIO] = convenio),
              FILTER(Dim_Unidades, Dim_Unidades[TIPO] = unidade),
              FILTER(_CALENDARIO, _CALENDARIO[Mes/Ano] = mes),
              ALL(Cirurgias)
          )
    - Qtd Partos:
      
          CALCULATE(
              SUM(Cirurgias[QTD]),
              Cirurgias[NR_PROCEDIMENTO] IN {"PARTO NORMAL", "PARTO COM VACUO EXTRATOR", "PARTO COM OU SEM EPISIOTOMIA", "PARTO CESARIANA + LAQUEADURA", "PARTO MULTIPLO (CADA UM SUBSEQUENTE AO INICIA", "PARTO A FORCEPS", "LAQUEADURA TUBARIA", "LAQUEADURA TUBARIA LAPAROSCOPICA", "CESARIANA"} ||
              Cirurgias[TIPO] in {"PARTOS", "PARTOS VARGINAIS"}
          )
