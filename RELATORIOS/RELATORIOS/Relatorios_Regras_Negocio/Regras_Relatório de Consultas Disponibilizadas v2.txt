Tabela: #MEDIDAS
  Medidas DAX (Regras de Negócio):
    - Qt Registros:
      COUNTROWS(fDisponibilizadas)
    - Qt Disponib.:
      SUM(fDisponibilizadas[QTE_DISPONIBILIZADA])
    - Qt Medicos:
      DISTINCTCOUNT( fDisponibilizadas[CD_MEDICO] )
    - Msg Boas Vindas:
      
      VAR vUsuario = LEFT(USERPRINCIPALNAME(), FIND("@", USERPRINCIPALNAME()) - 1)
      VAR vHorarioAtual = NOW() - TIME(3, 0, 0) // Subtrai 3 horas para ajustar ao fuso horário de Brasília (UTC -3)
      VAR vSaudacao = 
          SWITCH(
              TRUE(),
              HOUR(vHorarioAtual) < 12, ", bom dia. Seja bem-vindo(a)!",
              HOUR(vHorarioAtual) < 18, ", boa tarde. Seja bem-vindo(a)!",
              HOUR(vHorarioAtual) <= 23, ", boa noite. Seja bem-vindo(a)!"
          )
      RETURN
      "Olá, " & vUsuario & vSaudacao
      
    - Qtde Disp Formatada:
      
      FORMAT([Qt Disponib.], "#,0") 
    - Atualizado em:
      
       VAR vDataAtualizacao = SELECTEDVALUE(DT_ATUALIZACAO[DT_ATUALIZACAO])
       RETURN
       "Atualizado em: " & vDataAtualizacao
    - Qtde Disp T0012:
      SUM(fDispT0012[Qtde Disponib.])
    - Qtde Disp Format2:
      
      FORMAT([Qtde Disp T0012], "#,0") 
    - Eixo Ajustado:
      [Qt Medicos] * 1.2
    - Qt Registros T0012:
      COUNTROWS(fDispT0012)
    - Cidade Filtrada:
      
      CALCULATE(
          VALUES(dCidade[COD_UNIDADE]),
          CROSSFILTER(dCidade[COD_UNIDADE], fDisponibilizadas[CD_LOCAL], BOTH)
      )
