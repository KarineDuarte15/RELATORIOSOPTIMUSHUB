Tabela: #MEDIDAS
  Medidas DAX (Regras de Negócio):
    - Qtde Atrasos:
      COUNTROWS(FATO_ATRASO_MEDICO)
    - Tempo Total Atraso:
      SUM(FATO_ATRASO_MEDICO[TEMPO ATRASO])
    - Qtde Atrasos Dinamico:
      
      CALCULATE(
          [Qtde Atrasos],
          FILTER(
              FATO_ATRASO_MEDICO,
              [Tempo Total Atraso]* 60 > [Valor Parâmetro]
      ))
    - Tempo Total Atraso Dinamico:
      
      CALCULATE(
          [Tempo Total Atraso],
          FILTER(
              FATO_ATRASO_MEDICO,
              [Tempo Total Atraso]* 60 > [Valor Parâmetro]
      ))
      
    - Média de Atrasos:
      
      VAR vHorasDecimal = DIVIDE([Tempo Total Atraso Dinamico] , [Qtde Atrasos Dinamico])
      VAR vHoras = INT(vHorasDecimal)
      VAR vMinutosDecimal = 60 * (vHorasDecimal - vHoras)
      VAR vMinutos = INT(vMinutosDecimal)
      VAR vSegundos = ROUND( 60 * (vMinutosDecimal - vMinutos), 0 )
      VAR vHH = IF(LEN(vHoras) = 1 , "0" & vHoras, vHoras)
      VAR vMM = IF(LEN(vMinutos) = 1 , "0" & vMinutos, vMinutos)
      VAR vSS = IF(LEN(vSegundos) = 1 , "0" & vSegundos, vSegundos)
      RETURN
      CONVERT(vHH&vMM&vSS, INTEGER)
      
    - Total Atraso Dinamico Formato hh:mm:ss:
      
      VAR vTempoAtraso = 
          CALCULATE(
              [Tempo Total Atraso],
              FILTER(
                  FATO_ATRASO_MEDICO,
                  [Tempo Total Atraso] * 60 >= [Valor Parâmetro]
              )
          )
      VAR vHoras = INT(vTempoAtraso)
      VAR vMinutosDecimal = 60 * (vTempoAtraso - vHoras)
      VAR vMinutos = INT(vMinutosDecimal)
      VAR vSegundos = ROUND( 60 * (vMinutosDecimal - vMinutos), 0 )
      /*
      VAR vHH = CONVERT(CONVERT(IF(LEN(vHoras) >= 1 , "0" & vHoras, vHoras),INTEGER),STRING)
      VAR vMM = CONVERT(IF(LEN(vMinutos) >= 1 , "0" & vMinutos, vMinutos),INTEGER)
      VAR vSS = CONVERT(IF(LEN(vSegundos) >= 1 , "0" & vSegundos, vSegundos), INTEGER)
      */
      RETURN
      VALUE(vHoras & vMinutos & vSegundos)
    - Data Atualização:
      "Atualizado em: " & SELECTEDVALUE('#ATUALIZAÇÃO'[#ATUALIZAÇÃO])
    - ##Msg Boas Vindas:
      
      VAR vUsuario = LEFT(USERPRINCIPALNAME(), FIND("@", USERPRINCIPALNAME()) - 1)
      VAR vHorarioAtual = NOW() - TIME(3, 0, 0) // Subtrai 3 horas para ajustar ao fuso horário de Brasília (UTC -3)
      VAR vSaudacao = 
          SWITCH(
              TRUE(),
              HOUR(vHorarioAtual) < 12, ", bom dia. Seja bem-vindo(a)!",
              HOUR(vHorarioAtual) < 18, ", boa tarde. Seja bem-vindo(a)!",
              HOUR(vHorarioAtual) <= 23, ", boa noite. Seja bem-vindo(a)!"
          )
      RETURN
      "Olá, " & vUsuario & vSaudacao
